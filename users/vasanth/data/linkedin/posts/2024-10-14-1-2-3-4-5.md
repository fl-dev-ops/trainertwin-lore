Given above tree make in-order traversal of it. The code is hardly 10 lines as given below, but most beginners fail to visualise it. Refer below diagram given by ChatGPT to learn recursion visualisation. 

function inorderTraversal(root: TreeNode | null): number[] {
  let result: number[] = [];
  function traverse(node: TreeNode | null) {
    if (!node) return;
    traverse(node.left); // Step 1: Visit left
    result.push(node.val); // Step 2: Visit current node
    traverse(node.right); // Step 3: Visit right
  }
  traverse(root);
  return result;
}
